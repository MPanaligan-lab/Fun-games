    <script>
        const apiKey = ""; // Provided at runtime
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreEl = document.getElementById('score');

        // Game Settings
        const GRAVITY = 0.6;
        const JUMP_FORCE = -14;
        const SPEED = 6;
        const GROUND_Y_OFFSET = 100;

        let gameActive = false;
        let score = 0;
        let scrollOffset = 0;
        let keys = {};

        const player = { x: 100, y: 0, w: 40, h: 50, vx: 0, vy: 0, grounded: false };

        let wordPool = [
            { text: 'fly', type: 'correct' }, { text: 'my', type: 'correct' },
            { text: 'sky', type: 'correct' }, { text: 'cry', type: 'correct' },
            { text: 'dry', type: 'correct' }, { text: 'try', type: 'correct' },
            { text: 'candy', type: 'wrong' }, { text: 'baby', type: 'wrong' },
            { text: 'happy', type: 'wrong' }, { text: 'sunny', type: 'wrong' }
        ];

        let platforms = [];
        let items = [];
        let particles = [];

        // --- GEMINI AI INTEGRATION ---

        async function fetchAIWords(theme) {
            const systemPrompt = `You are an educational assistant for UFLI Lesson 73. 
            The goal is to teach Grade 4 students when terminal 'y' makes a long /i/ sound (like fly, sky).
            Task: Provide a JSON object with two arrays: 
            1. "correct": 6 words ending in 'y' that make the long /i/ sound, inspired by the theme "${theme}".
            2. "wrong": 6 words ending in 'y' that make the long /e/ sound (like candy), inspired by the theme "${theme}".
            Words must be age-appropriate for 4th grade.`;

            const userQuery = `Generate words for theme: ${theme}`;

            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: userQuery }] }],
                        systemInstruction: { parts: [{ text: systemPrompt }] },
                        generationConfig: { responseMimeType: "application/json" }
                    })
                });
                const data = await response.json();
                const result = JSON.parse(data.candidates[0].content.parts[0].text);
                
                return [
                    ...result.correct.map(w => ({ text: w, type: 'correct' })),
                    ...result.wrong.map(w => ({ text: w, type: 'wrong' }))
                ];
            } catch (err) {
                console.error(err);
                return wordPool; // Fallback
            }
        }

        async function speakWord(text) {
            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: `Say clearly: ${text}` }] }],
                        generationConfig: { 
                            responseModalities: ["AUDIO"],
                            speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: "Puck" } } }
                        }
                    })
                });
                const data = await response.json();
                const audioBase64 = data.candidates[0].content.parts[0].inlineData.data;
                const audioBlob = base64ToWav(audioBase64, 24000); // 24kHz typical for this model
                const url = URL.createObjectURL(audioBlob);
                const audio = new Audio(url);
                audio.play();
            } catch (err) {
                console.error("TTS Error", err);
            }
        }

        function base64ToWav(base64, sampleRate) {
            const binaryString = atob(base64);
            const len = binaryString.length;
            const bytes = new Int16Array(len / 2);
            for (let i = 0; i < len; i += 2) {
                bytes[i / 2] = (binaryString.charCodeAt(i + 1) << 8) | binaryString.charCodeAt(i);
            }
            const buffer = new ArrayBuffer(44 + bytes.length * 2);
            const view = new DataView(buffer);
            const writeString = (offset, string) => {
                for (let i = 0; i < string.length; i++) view.setUint8(offset + i, string.charCodeAt(i));
            };
            writeString(0, 'RIFF');
            view.setUint32(4, 36 + bytes.length * 2, true);
            writeString(8, 'WAVE');
            writeString(12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true);
            view.setUint16(22, 1, true);
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeString(36, 'data');
            view.setUint32(40, bytes.length * 2, true);
            const offset = 44;
            for (let i = 0; i < bytes.length; i++) view.setInt16(offset + i * 2, bytes[i], true);
            return new Blob([buffer], { type: 'audio/wav' });
        }

        async function generateAILevel() {
            const theme = document.getElementById('ai-theme').value || "Fun Adventure";
            document.getElementById('ai-loading-screen').classList.remove('hidden');
            wordPool = await fetchAIWords(theme);
            document.getElementById('ai-loading-screen').classList.add('hidden');
            startGame();
        }

        // --- GAME ENGINE ---

        function initLevel() {
            const h = canvas.height;
            platforms = [
                { x: 0, y: h - GROUND_Y_OFFSET, w: 1000, h: GROUND_Y_OFFSET },
                { x: 1150, y: h - GROUND_Y_OFFSET, w: 1500, h: GROUND_Y_OFFSET },
                { x: 400, y: h - 250, w: 200, h: 40 },
                { x: 750, y: h - 400, w: 200, h: 40 },
                { x: 1300, y: h - 250, w: 300, h: 40 },
                { x: 1700, y: h - 450, w: 200, h: 40 },
                { x: 2100, y: h - 300, w: 400, h: 40 },
                { x: 2650, y: h - GROUND_Y_OFFSET, w: 500, h: GROUND_Y_OFFSET }
            ];

            items = [];
            for (let i = 0; i < 25; i++) {
                const word = wordPool[Math.floor(Math.random() * wordPool.length)];
                items.push({
                    x: 500 + (i * 250),
                    y: h - 200 - (Math.random() * 350),
                    w: 120,
                    h: 45,
                    text: word.text,
                    type: word.type,
                    collected: false,
                    seed: Math.random() * 100
                });
            }
        }

        function drawPlayer() {
            ctx.fillStyle = '#e74c3c';
            ctx.fillRect(player.x - scrollOffset, player.y, player.w, player.h);
            ctx.fillStyle = '#f39c12';
            ctx.fillRect(player.x - scrollOffset, player.y + 20, player.w, 15);
            ctx.fillStyle = '#2c3e50';
            ctx.fillRect(player.x - scrollOffset + 25, player.y + 10, 8, 8);
            ctx.fillStyle = '#c0392b';
            ctx.fillRect(player.x - scrollOffset - 5, player.y - 5, player.w + 10, 10);
        }

        function drawPlatform(p) {
            ctx.fillStyle = '#8e44ad';
            ctx.fillRect(p.x - scrollOffset, p.y, p.w, p.h);
            ctx.fillStyle = '#9b59b6';
            ctx.fillRect(p.x - scrollOffset, p.y, p.w, 8);
        }

        function drawItem(it) {
            if (it.collected) return;
            const floatY = Math.sin(Date.now() / 300 + it.seed) * 10;
            ctx.fillStyle = it.type === 'correct' ? '#fff' : '#ff9999';
            ctx.strokeStyle = '#2c3e50';
            ctx.lineWidth = 3;
            ctx.beginPath();
            ctx.roundRect(it.x - scrollOffset, it.y + floatY, it.w, it.h, 10);
            ctx.fill();
            ctx.stroke();
            ctx.fillStyle = '#2c3e50';
            ctx.font = '10px "Press Start 2P"';
            ctx.textAlign = 'center';
            ctx.fillText(it.text, it.x - scrollOffset + it.w / 2, it.y + floatY + 28);
        }

        function update() {
            if (!gameActive) return;

            if (keys['ArrowRight']) player.vx = SPEED;
            else if (keys['ArrowLeft']) player.vx = -SPEED;
            else player.vx *= 0.8;

            player.vy += GRAVITY;
            player.x += player.vx;
            player.y += player.vy;

            if (player.x < 0) player.x = 0;

            player.grounded = false;
            platforms.forEach(p => {
                if (player.x + player.w > p.x && player.x < p.x + p.w) {
                    if (player.y + player.h > p.y && player.y + player.h < p.y + player.vy + 10) {
                        player.y = p.y - player.h;
                        player.vy = 0;
                        player.grounded = true;
                    }
                }
            });

            if (player.x > canvas.width / 2) {
                scrollOffset = player.x - canvas.width / 2;
            }

            items.forEach(it => {
                if (!it.collected && 
                    player.x < it.x + it.w && 
                    player.x + player.w > it.x &&
                    player.y < it.y + it.h &&
                    player.y + player.h > it.y) {
                    
                    it.collected = true;
                    speakWord(it.text); // ✨ TTS Integration
                    
                    if (it.type === 'correct') {
                        score += 100;
                        createParticles(it.x + it.w/2, it.y + it.h/2, '#f1c40f');
                    } else {
                        score = Math.max(0, score - 150);
                        createParticles(it.x + it.w/2, it.y + it.h/2, '#e74c3c');
                    }
                    scoreEl.innerText = score;
                }
            });

            if (player.y > canvas.height) endGame(false);
            if (player.x > 2800) endGame(true);

            particles.forEach((p, i) => {
                p.x += p.vx; p.y += p.vy; p.a -= 0.02;
                if (p.a <= 0) particles.splice(i, 1);
            });
        }

        function createParticles(x, y, color) {
            for (let i = 0; i < 10; i++) {
                particles.push({
                    x, y, color, a: 1,
                    vx: (Math.random() - 0.5) * 8,
                    vy: (Math.random() - 0.5) * 8
                });
            }
        }

        function draw() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            platforms.forEach(drawPlatform);
            items.forEach(drawItem);
            particles.forEach(p => {
                ctx.globalAlpha = p.a;
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x - scrollOffset, p.y, 6, 6);
            });
            ctx.globalAlpha = 1;
            drawPlayer();

            ctx.fillStyle = '#2c3e50';
            ctx.fillRect(2850 - scrollOffset, canvas.height - 600, 10, 500);
            ctx.fillStyle = '#f1c40f';
            ctx.beginPath();
            ctx.moveTo(2860 - scrollOffset, canvas.height - 600);
            ctx.lineTo(2920 - scrollOffset, canvas.height - 560);
            ctx.lineTo(2860 - scrollOffset, canvas.height - 520);
            ctx.fill();

            if (gameActive) {
                requestAnimationFrame(draw);
                update();
            }
        }

        function startGame() {
            document.getElementById('start-screen').classList.add('hidden');
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            player.y = canvas.height - 200;
            player.x = 100;
            score = 0;
            scrollOffset = 0;
            scoreEl.innerText = score;
            initLevel();
            gameActive = true;
            draw();
        }

        function endGame(win) {
            gameActive = false;
            if (win) {
                document.getElementById('win-screen').classList.remove('hidden');
                document.getElementById('final-score-win').innerText = "Score: " + score;
            } else {
                document.getElementById('game-over-screen').classList.remove('hidden');
            }
        }

        window.addEventListener('keydown', e => {
            keys[e.key] = true;
            if ((e.key === 'ArrowUp' || e.key === ' ') && player.grounded) {
                player.vy = JUMP_FORCE;
                player.grounded = false;
            }
        });
        window.addEventListener('keyup', e => keys[e.key] = false);
        window.addEventListener('resize', () => {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        });
    </script>
